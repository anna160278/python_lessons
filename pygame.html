🎮 Что такое pygame и первое окно

Pygame — это библиотека Python для создания 2D-игр. С её помощью можно рисовать графику, воспроизводить звуки и обрабатывать нажатия клавиш и мыши.

Установка pygame

Pygame не входит в стандартную библиотеку Python — его нужно установить отдельно. Открой терминал (командную строку) и напиши:

# В терминале (не в редакторе Python!): pip install pygame # После установки проверяем в Python: import pygame print(pygame.ver) # Должна появиться версия, например 2.5.0

Импорт: import pygame as pg

В большинстве игр pygame импортируют с псевдонимом pg. Это общепринятый стандарт, который делает код короче и чище.

import pygame as pg # Теперь вместо pygame.init() мы пишем pg.init() # Вместо pygame.draw.rect() мы пишем pg.draw.rect()

Первое окно и игровой цикл

import pygame as pg # 1. Инициализируем все модули pygame (обязательно в начале!) pg.init() # 2. Создаём игровое окно размером 600 на 400 пикселей screen = pg.display.set_mode((600, 400)) # 3. Задаём заголовок окна pg.display.set_caption("Моя первая игра") # 4. Флаг для управления игровым циклом running = True # 5. Игровой цикл: работает, пока running == True while running: # Обрабатываем все события (клавиши, мышь, закрытие окна) for event in pg.event.get(): # Если пользователь нажал на крестик закрытия окна if event.type == pg.QUIT: running = False # Останавливаем цикл # 6. Заливаем экран тёмно-серым цветом screen.fill((30, 30, 30)) # 7. Обновляем содержимое всего экрана pg.display.flip() # показываем всё, что нарисовали # 8. Корректно закрываем pygame после выхода из цикла pg.quit()

🎨 FPS и рисование примитивов

Чтобы игра работала плавно и с одинаковой скоростью на разных компьютерах, нужно ограничить количество кадров в секунду (FPS).

Clock и ограничение FPS

import pygame as pg pg.init() screen = pg.display.set_mode((600, 400)) # Создаём объект Clock для контроля времени clock = pg.time.Clock() FPS = 60 # Цель: 60 кадров в секунду running = True while running: for event in pg.event.get(): if event.type == pg.QUIT: running = False screen.fill((30, 30, 30)) pg.display.flip() # Заставляем цикл ждать, clock.tick(FPS) # чтобы не превышать 60 кадров в секунду pg.quit()

Рисование фигур

Все функции рисования находятся в модуле pg.draw. Цвета задаются кортежем RGB: (Red, Green, Blue) от 0 до 255.

# Прямоугольник: (экран, цвет, (x, y, ширина, высота)) pg.draw.rect(screen, (255, 0, 0), (50, 50, 100, 60)) # Круг: (экран, цвет, (x центра, y центра), радиус) pg.draw.circle(screen, (0, 255, 0), (300, 200), 40) # Линия: (экран, цвет, (x1, y1), (x2, y2), толщина) pg.draw.line(screen, (0, 0, 255), (0, 0), (600, 400), 3)

📐 Атрибуты объекта Rect

pg.Rect — это прямоугольник, который хранит позицию и размер объекта. У него есть множество "виртуальных" атрибутов, которые позволяют легко перемещать и выравнивать объекты без сложной математики.

# Создаём прямоугольник: (x, y, ширина, высота) small_rect = pg.Rect(0, 0, 50, 50) # --- Основные координаты --- small_rect.x, small_rect.y # Верхний левый угол small_rect.top, small_rect.left # Верхняя и левая границы small_rect.bottom, small_rect.right # Нижняя и правая границы # --- Угловые точки (возвращают кортеж (x, y)) --- small_rect.topleft # Верхний левый угол small_rect.bottomright # Нижний правый угол # --- Середины сторон и центр --- small_rect.midtop # Середина верхней стороны small_rect.center # (x, y) центра прямоугольника small_rect.centerx # Только X центра # --- Размеры --- small_rect.size # Кортеж (width, height) small_rect.width # или small_rect.w small_rect.height # или small_rect.h

Пример: центрирование объекта

size = (600, 400) # Размер нашего игрового окна small_rect = pg.Rect(0, 0, 50, 50) # Ставим прямоугольник ровно в центр экрана одной строкой! small_rect.center = (size[0] // 2, size[1] // 2) # Прижимаем правый край прямоугольника к правому краю окна small_rect.right = size[0]

🔊 Звуки и музыка

В pygame есть модуль mixer. Он делит аудио на два типа: короткие звуковые эффекты (Sound) и длинная фоновая музыка (music).

1. Короткие звуки (Sound) - подходят для выстрелов, прыжков, подбора монет (можно накладывать друг на друга). Форматы: WAV, OGG

2. Фоновая музыка (music) - подходит для саундтреков. Проигрывается только один трек за раз. Форматы: MP3, OGG

import pygame as pg pg.init() pg.mixer.init() # Явно инициализируем звуковой модуль shoot_sound = pg.mixer.Sound("shoot.wav") shoot_sound.play() # Воспроизвести звук shoot_sound.set_volume(0.5) # Громкость от 0.0 до 1.0 pg.mixer.music.load("background.mp3") pg.mixer.music.play(-1) # -1 означает бесконечное повторение по кругу pg.mixer.music.set_volume(0.3) # Громкость музыки (тише, чем звуки) # Управление музыкой pg.mixer.music.pause() # Поставить на паузу pg.mixer.music.unpause() # Продолжить воспроизведение pg.mixer.music.stop() # Полностью остановить

👥 Группы спрайтов

Когда объектов много, управлять ими по отдельности неудобно. Класс pg.sprite.Sprite и pg.sprite.Group решают эту проблему.

Создание своего спрайта

class Player(pg.sprite.Sprite): def __init__(self, x, y): super().__init__() # ❗ вызываем конструктор родителя self.image = pg.image.load("player.png") # 1. Загружаем картинку self.image = pg.transform.scale(self.image, (50, 50)) # Задаём размер self.rect = self.image.get_rect(topleft=(x, y)) # 2. Получаем прямоугольник и задаём координаты # ❗ self.image и self.rect # - два ОБЯЗАТЕЛЬНЫХ атрибута для любого спрайта def update(self): # Сюда пишем логику движения. Вызывается в игровом цикле keys = pg.key.get_pressed() if keys[pg.K_LEFT]: self.rect.x -= 5

Использование групп

# Создаём пустые группы all_sprites = pg.sprite.Group() # Для отрисовки всех спрайтов одной строкой enemies = pg.sprite.Group() # Для удобной проверки столкновений # Создаём игрока и добавляем его в группу player = Player(100, 100) all_sprites.add(player) while running: # 1. Обновляем логику ВСЕХ спрайтов в группе all_sprites.update() # вызываем их метод update # 2. Очищаем экран screen.fill((30, 30, 30)) # заливка фона # 3. Отрисовываем ВСЕ спрайты группы all_sprites.draw(screen) # используя self.image и self.rect pg.display.flip()

💥 Столкновения: от Rect до Масок

В играх постоянно нужно проверять, коснулись ли объекты друг друга.
Начнём с самого простого и дойдём до пиксельной точности.

1. Простое столкновение: colliderect

Самый быстрый способ. Проверяет пересечение двух прямоугольников.
Минус: не учитывает прозрачные области картинки.

# Создаём прямоугольники для игрока и врага player_rect = pg.Rect(x, y, 30, 30) enemy_rect = pg.Rect(ex, ey, 40, 40) # Проверяем, пересекаются ли они if player_rect.colliderect(enemy_rect): print("Столкновение!")

2. Спрайты и spritecollide

Когда врагов много, удобнее использовать группы. Метод проверяет столкновение одного спрайта с группой спрайтов.

# starship — это один объект (наш корабль) # meteorites — это группа pg.sprite.Group() # содержащая много метеоритов hits = pg.sprite.spritecollide(starship, meteorites, True) # Третий аргумент True означает: # удалить метеорит из группы при столкновении if hits: # hits — это список всех метеоритов, в которые мы врезались lives -= len(hits) # Отнимаем жизни за каждое попадание

3. Пиксельно-точное столкновение: collide_mask

Если у корабля круглая форма, а прямоугольник квадратный, colliderect даст ложное срабатывание в углах.
Решение — маски! Используем тот же spritecollide, но добавляем 4-й аргумент. pg.sprite.collide_mask заставит проверять только непрозрачные пиксели!

hits = pg.sprite.spritecollide(starship, mice, True, pg.sprite.collide_mask) if hits: print("Поймали мышь точно по контуру!")

4. Столкновение группы с группой: groupcollide

Идеально для шутеров: проверяем, попали ли любые пули в любых врагов.

# lasers — группа пуль, mice — группа врагов # Первый True: удалить пулю при попадании. # Второй True: удалить врага при попадании. hits = pg.sprite.groupcollide(lasers, mice, True, True) # hits — это словарь вида: # {пуля: [список поражённых этой пулей мышей]} for laser, hit_mice in hits.items(): # Начисляем очки за каждую уничтоженную мышь score += len(hit_mice) * 10

🐍 Мини-игра: Змейка

Соберём все знания вместе! Управление стрелками, еда появляется случайно, змейка растёт при поедании.

import pygame as pg import random pg.init() # --- Настройки игры --- CELL = 20 # Размер одной клетки в пикселях COLS, ROWS = 30, 20 # Поле 30x20 клеток screen = pg.display.set_mode((COLS * CELL, ROWS * CELL)) clock = pg.time.Clock() # --- Начальное состояние --- snake = [(15, 10)] # Змейка — это список координат её сегментов dx, dy = 1, 0 # Начальное направление: вправо (x+1, y+0) food = (random.randint(0, COLS-1), random.randint(0, ROWS-1)) running = True while running: # 1. Обработка событий for event in pg.event.get(): if event.type == pg.QUIT: running = False elif event.type == pg.KEYDOWN: # Запрещаем разворот на 180 градусов (dy != 1 означает, что мы не движемся вниз) if event.key == pg.K_UP and dy != 1: dx, dy = 0, -1 elif event.key == pg.K_DOWN and dy != -1: dx, dy = 0, 1 elif event.key == pg.K_LEFT and dx != 1: dx, dy = -1, 0 elif event.key == pg.K_RIGHT and dx != -1: dx, dy = 1, 0 # 2. Двигаем змейку: создаём новую голову на основе текущей головы + направление head = (snake[0][0] + dx, snake[0][1] + dy) # 3. Проверка проигрыша: удар о стену или о собственный хвост if head[0] < 0 or head[0] >= COLS or head[1] < 0 or head[1] >= ROWS or head in snake: running = False # 4. Добавляем новую голову в начало списка snake.insert(0, head) # 5. Логика еды if head == food: # Если съели: генерируем новую еду, хвост НЕ удаляем (змейка выросла) food = (random.randint(0, COLS-1), random.randint(0, ROWS-1)) else: # Если не съели: удаляем последний элемент списка (хвост), чтобы длина не менялась snake.pop() # 6. Отрисовка screen.fill((30, 30, 30)) # Рисуем каждый сегмент змейки for segment in snake: pg.draw.rect(screen, (0, 255, 0), (segment[0] * CELL, segment[1] * CELL, CELL, CELL)) # Рисуем еду pg.draw.rect(screen, (255, 0, 0), (food[0] * CELL, food[1] * CELL, CELL, CELL)) pg.display.flip() clock.tick(10) # Скорость игры: 10 кадров (шагов) в секунду pg.quit()
Как работает змейка:
